\newcommand{\cjksq}[4]{% #1 rozmiar boxu  #2 rozmiar znaku  #3 wariant(D/L)  #4 znak
  \ifstrequal{#3}{D}%
  {\tikz[baseline=-3.4pt]{\node[draw=goldlineD, fill=vermilion!12!inkbg,
      rounded corners=1.6pt, inner sep=0pt, minimum size=#1, line width=0.5pt]
      {\cjkfont\bfseries\fontsize{#2}{#2}\selectfont\color{vermilion}#4};}}%
  {\tikz[baseline=-3.8pt]{\node[draw=goldlineL, fill=lvermilion!9!paper,
      rounded corners=1.8pt, inner sep=0pt, minimum size=#1, line width=0.5pt]
      {\cjkfont\bfseries\fontsize{#2}{#2}\selectfont\color{lvermilion}#4};}}}
